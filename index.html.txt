<script>
    const checkboxes = document.querySelectorAll('.task-check');
    const progressBar = document.getElementById('progress-bar');
    const progressPercent = document.getElementById('progress-percent');
    const footerStatus = document.getElementById('footer-status-text');
    const resetButton = document.getElementById('reset-all');

    function updateProgress() {
        const total = checkboxes.length;
        const checked = document.querySelectorAll('.task-check:checked').length;
        const percentage = Math.round((checked / total) * 100);

        progressBar.style.width = percentage + '%';
        progressPercent.innerText = percentage + '% concluído';
        footerStatus.innerText = `Você completou ${checked} de ${total} ações!`;

        if (percentage === 100) {
            progressBar.style.backgroundColor = 'var(--success)';
        } else {
            progressBar.style.backgroundColor = 'var(--primary-light)';
        }

        // Save to localStorage
        const state = Array.from(checkboxes).map(cb => cb.checked);
        localStorage.setItem('ebook_progress_state', JSON.stringify(state));
    }

    // Load from localStorage
    function loadProgress() {
        const savedState = localStorage.getItem('ebook_progress_state');
        if (savedState) {
            const state = JSON.parse(savedState);
            checkboxes.forEach((cb, index) => {
                cb.checked = state[index] || false;
            });
        }
        updateProgress();
    }

    checkboxes.forEach(cb => {
        cb.addEventListener('change', updateProgress);
    });

    resetButton.addEventListener('click', () => {
        if (confirm('Deseja desmarcar todas as ações?')) {
            checkboxes.forEach(cb => cb.checked = false);
            updateProgress();
        }
    });

    // Initialize
    loadProgress();
</script>
